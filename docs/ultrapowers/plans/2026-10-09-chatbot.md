# Chatbot OfficeTure · plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use ultrapowers:subagent-driven-development (recommended) or ultrapowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir el asistente de ventas de OfficeTure para WhatsApp e Instagram (Claude + Supabase + funciones de Vercel), probado de punta a punta con el número de prueba de Meta y un simulador de conversaciones.

**Architecture:** Los webhooks de Meta guardan cada evento en Supabase y responden 200; el procesamiento corre después con `waitUntil`. Cada mensaje del cliente mueve `responder_en` de su conversación 4 s al futuro; `pg_cron` llama cada 5 s a un despachador que toma las conversaciones vencidas con `skip locked` y ejecuta el turno: Claude con herramientas sobre el inventario, envío por el adaptador del canal y, si aplica, escalación a Bryan. Un cron de Vercel por minuto despacha seguimientos, vence modos humanos y reenvía alertas. Toda la lógica depende de una interfaz `Repo` con dos implementaciones: Supabase (producción) y memoria (pruebas y simulador).

**Tech Stack:** Node 24, TypeScript, Vercel Functions (sin framework), Supabase (Postgres, `pg_cron`, `pg_net`, Storage, Vault), `@anthropic-ai/sdk@0.133.0` con `claude-sonnet-5-5`, WhatsApp Cloud API y Instagram Messaging API por `fetch`, `googleapis@185.0.0`, transcripción con `gpt-4o-mini-transcribe`, vitest, biome.

**Fuentes:** spec `docs/ultrapowers/specs/2026-10-09-chatbot-design.md`, brief `docs/ultrapowers/research/2026-10-09-chatbot-research-brief.md` y sus partes `research/partes/01` a `05`.

```ts
  RESEND_API_KEY: z.string().default(""),
  RESEND_FROM: z.string().default(""),

export type Config = z.infer<typeof Esquema>;

export function cargarConfig(env: Record<string, string | undefined> = process.env): Config {
  return Esquema.parse(env);
}

export interface Dueno {
  nombre: string;
  telefono: string;
}

export function duenos(c: Config): Dueno[] {
  return c.DUENOS.split(",")
    .map((par) => par.trim())
    .filter(Boolean)
    .map((par) => {
      const [nombre = "", telefono = ""] = par.split(":");
      return { nombre: nombre.trim(), telefono: telefono.trim() };
    });
}

export const ultimos10 = (numero: string): string => numero.replace(/\D/g, "").slice(-10);

export function esDueno(c: Config, numero: string): boolean {
  const n = ultimos10(numero);
  return duenos(c).some((d) => ultimos10(d.telefono) === n);
}
```

```ts
/** Monterrey: UTC-6 todo el año desde 2022. */
const OFFSET_H = -6;
const HORA_MS = 3_600_000;

export interface Local {
  dia: number; // 0 domingo … 6 sábado
  hora: number; // decimal, 10.5 = 10:30
  anio: number;
  mes: number; // 0-11
  fecha: number;
}

export function local(d: Date): Local {
  const t = new Date(d.getTime() + OFFSET_H * HORA_MS);
  return {
    dia: t.getUTCDay(),
    hora: t.getUTCHours() + t.getUTCMinutes() / 60,
    anio: t.getUTCFullYear(),
    mes: t.getUTCMonth(),
    fecha: t.getUTCDate(),
  };
}

export function aLocal(anio: number, mes: number, fecha: number, hora: number, minuto = 0): Date {
  return new Date(Date.UTC(anio, mes, fecha, hora - OFFSET_H, minuto));
}

/** Lunes a viernes 9 a 19, sábado 10 a 19. Pendiente de confirmar con el cliente. */
export function enHorario(d: Date): boolean {
  const l = local(d);
  if (l.dia === 0) return false;
  const abre = l.dia === 6 ? 10 : 9;
  return l.hora >= abre && l.hora < 19;
}

/** Día hábil = lunes a sábado. Devuelve el siguiente día hábil (después de hoy) a la hora dada. */
export function siguienteDiaHabilA(d: Date, hora: number): Date {
  const l = local(d);
  let candidato = aLocal(l.anio, l.mes, l.fecha + 1, hora);
  while (local(candidato).dia === 0) {
    const c = local(candidato);
    candidato = aLocal(c.anio, c.mes, c.fecha + 1, hora);
  }
  return candidato;
}

/** Hoy a la hora dada si aún no pasa; si no, mañana a esa hora. */
export function proximaA(d: Date, hora: number): Date {
  const l = local(d);
  const hoy = aLocal(l.anio, l.mes, l.fecha, hora);
  return hoy.getTime() > d.getTime() ? hoy : aLocal(l.anio, l.mes, l.fecha + 1, hora);
}

/** Los seguimientos solo salen de 9:00 a 20:00 hora local. */
export function ajustarVentanaToques(d: Date): Date {
  const l = local(d);
  if (l.hora >= 9 && l.hora < 20) return d;
  return l.hora < 9 ? aLocal(l.anio, l.mes, l.fecha, 9) : aLocal(l.anio, l.mes, l.fecha + 1, 9);
}

export const horas = (n: number): number => n * HORA_MS;

export function describirLocal(d: Date): string {
  const dias = ["domingo", "lunes", "martes", "miércoles", "jueves", "viernes", "sábado"];
  const meses = ["enero", "febrero", "marzo", "abril", "mayo", "junio", "julio", "agosto", "septiembre", "octubre", "noviembre", "diciembre"];
  const l = local(d);
  const hh = Math.floor(l.hora).toString().padStart(2, "0");
  const mm = Math.round((l.hora % 1) * 60).toString().padStart(2, "0");
  return `${dias[l.dia]} ${l.fecha} de ${meses[l.mes]}, ${hh}:${mm}`;
}
```

```ts
import { createHmac, timingSafeEqual } from "node:crypto";

/** Valida X-Hub-Signature-256 sobre el cuerpo CRUDO (nunca sobre JSON re-serializado). */
export function firmaValida(cuerpoCrudo: string, encabezado: string | null, secreto: string): boolean {
  if (!encabezado?.startsWith("sha256=")) return false;
  const recibido = Buffer.from(encabezado.slice(7), "utf8");
  const esperado = Buffer.from(createHmac("sha256", secreto).update(cuerpoCrudo, "utf8").digest("hex"), "utf8");
  return recibido.length === esperado.length && timingSafeEqual(recibido, esperado);
}
```

```ts
import type { EventoCanal, MensajeEntrante, ReferenciaAnuncio } from "./tipos.js";

interface WaMensaje {
  from: string;
  id: string;
  timestamp: string;
  type: string;
  text?: { body: string };
  image?: { id: string; mime_type?: string; caption?: string };
  audio?: { id: string; mime_type?: string };
  location?: { latitude: number; longitude: number; name?: string; address?: string };
  interactive?: { button_reply?: { id: string; title: string }; list_reply?: { id: string; title: string } };
  button?: { text: string };
  referral?: { source_id?: string; headline?: string; body?: string; source_url?: string; ctwa_clid?: string };
  context?: { id?: string };
}

interface WaEco {
  from: string;
  to: string;
  id: string;
  timestamp: string;
  type: string;
  text?: { body: string };
}

interface WaValor {
  contacts?: { wa_id: string; profile?: { name?: string } }[];
  messages?: WaMensaje[];
  message_echoes?: WaEco[];
}

const iso = (ts: string): string => new Date(Number(ts) * 1000).toISOString();

function referencia(r: WaMensaje["referral"]): ReferenciaAnuncio | undefined {
  if (!r) return undefined;
  const ref: ReferenciaAnuncio = {};
  if (r.source_id) ref.id = r.source_id;
  if (r.headline) ref.titulo = r.headline;
  if (r.body) ref.cuerpo = r.body;
  if (r.source_url) ref.url = r.source_url;
  if (r.ctwa_clid) ref.ctwaClid = r.ctwa_clid;
  return ref;
}

function mensaje(m: WaMensaje, nombre: string | undefined): MensajeEntrante {
  const base: MensajeEntrante = { canal: "whatsapp", idExterno: m.id, de: m.from, tipo: "otro", fecha: iso(m.timestamp) };
  if (nombre) base.nombre = nombre;
  const ref = referencia(m.referral);
  if (ref) base.referencia = ref;
  if (m.context?.id) base.citaId = m.context.id;
  switch (m.type) {
    case "text":
      return { ...base, tipo: "texto", texto: m.text?.body ?? "" };
    case "image": {
      const r: MensajeEntrante = { ...base, tipo: "imagen", mediaId: m.image?.id ?? "" };
      if (m.image?.mime_type) r.mime = m.image.mime_type;
      if (m.image?.caption) r.texto = m.image.caption;
      return r;
    }
    case "audio": {
      const r: MensajeEntrante = { ...base, tipo: "audio", mediaId: m.audio?.id ?? "" };
      if (m.audio?.mime_type) r.mime = m.audio.mime_type;
      return r;
    }
    case "location": {
      const l = m.location;
      const lugar = [l?.name, l?.address].filter(Boolean).join(", ");
      return { ...base, tipo: "ubicacion", texto: `[ubicación] ${lugar} (${l?.latitude}, ${l?.longitude})`.replace("  ", " ") };
    }
    case "interactive": {
      const r = m.interactive?.button_reply ?? m.interactive?.list_reply;
      return { ...base, tipo: "boton", texto: r?.title ?? "" };
    }
    case "button":
      return { ...base, tipo: "boton", texto: m.button?.text ?? "" };
    default:
      return { ...base, texto: `[mensaje de tipo ${m.type}]` };
  }
}

export function parsearWhatsApp(payload: unknown): EventoCanal[] {
  const p = payload as { object?: string; entry?: { changes?: { field: string; value: WaValor }[] }[] } | null;
  if (!p || p.object !== "whatsapp_business_account") return [];
  const eventos: EventoCanal[] = [];
  for (const entrada of p.entry ?? []) {
    for (const cambio of entrada.changes ?? []) {
      const v = cambio.value ?? {};
      if (cambio.field === "messages") {
        const nombres = new Map((v.contacts ?? []).map((c) => [c.wa_id, c.profile?.name]));
        for (const m of v.messages ?? []) eventos.push({ clase: "mensaje", m: mensaje(m, nombres.get(m.from)) });
      }
      if (cambio.field === "smb_message_echoes") {
        for (const e of v.message_echoes ?? []) {
          const eco = { canal: "whatsapp" as const, idExterno: e.id, para: e.to, fecha: iso(e.timestamp) };
          eventos.push({ clase: "eco", e: e.text?.body ? { ...eco, texto: e.text.body } : eco });
        }
      }
    }
  }
  return eventos;
}
```

```ts
import type { EventoCanal, MensajeEntrante, ReferenciaAnuncio } from "./tipos.js";

interface IgReferral {
  ad_id?: string;
  ref?: string;
  source?: string;
  ads_context_data?: { ad_title?: string };
}

interface IgMensaje {
  mid: string;
  text?: string;
  is_echo?: boolean;
  is_deleted?: boolean;
  attachments?: { type: string; payload?: { url?: string } }[];
  reply_to?: { story?: { id: string; url?: string } };
  referral?: IgReferral;
}

interface IgEvento {
  sender: { id: string };
  recipient: { id: string };
  timestamp: number;
  message?: IgMensaje;
  referral?: IgReferral;
}

function referencia(r: IgReferral | undefined): ReferenciaAnuncio | undefined {
  if (!r) return undefined;
  const ref: ReferenciaAnuncio = {};
  if (r.ad_id) ref.id = r.ad_id;
  if (r.ads_context_data?.ad_title) ref.titulo = r.ads_context_data.ad_title;
  return Object.keys(ref).length > 0 ? ref : undefined;
}

function mensaje(ev: IgEvento, m: IgMensaje): MensajeEntrante {
  const base: MensajeEntrante = { canal: "instagram", idExterno: m.mid, de: ev.sender.id, tipo: "otro", fecha: new Date(ev.timestamp).toISOString() };
  const ref = referencia(m.referral ?? ev.referral);
  if (ref) base.referencia = ref;
  const adjunto = m.attachments?.[0];
  if (adjunto?.type === "image" && adjunto.payload?.url) return { ...base, tipo: "imagen", mediaUrl: adjunto.payload.url, mime: "image/jpeg" };
  if (adjunto?.type === "audio" && adjunto.payload?.url) return { ...base, tipo: "audio", mediaUrl: adjunto.payload.url, mime: "audio/mp4" };
  if (m.text !== undefined) {
    const prefijo = m.reply_to?.story ? "[respondió a una historia] " : "";
    return { ...base, tipo: "texto", texto: `${prefijo}${m.text}` };
  }
  return { ...base, texto: `[adjunto de tipo ${adjunto?.type ?? "desconocido"}]` };
}

export function parsearInstagram(payload: unknown, igUserId: string): EventoCanal[] {
  const p = payload as { object?: string; entry?: { messaging?: IgEvento[] }[] } | null;
  if (!p || p.object !== "instagram") return [];
  const eventos: EventoCanal[] = [];
  for (const entrada of p.entry ?? []) {
    for (const ev of entrada.messaging ?? []) {
      const m = ev.message;
      if (!m || m.is_deleted) continue;
      if (m.is_echo) {
        const eco = { canal: "instagram" as const, idExterno: m.mid, para: ev.recipient.id, fecha: new Date(ev.timestamp).toISOString() };
        eventos.push({ clase: "eco", e: m.text ? { ...eco, texto: m.text } : eco });
        continue;
      }
      if (ev.sender.id === igUserId) continue;
      eventos.push({ clase: "mensaje", m: mensaje(ev, m) });
    }
  }
  return eventos;
}
```

```ts
export type Categoria = "escritorio" | "archivero" | "silla";
export type Uso = "colaboradores" | "direccion" | "recepcion" | "administracion" | "visitas" | "capacitacion" | "archivo";
export const USOS: Uso[] = ["colaboradores", "direccion", "recepcion", "administracion", "visitas", "capacitacion", "archivo"];

export interface Producto {
  slug: string;
  categoria: Categoria;
  subcategoria: string;
  nombre: string;
  precio: number; // MXN sin IVA
  stock: "alto" | "medio" | "bajo" | "proveedor";
  prioridad: "alta" | "media" | "baja";
  uso: Uso[];
  respuestaBase?: string;
  medidas?: string;
  colores?: string;
  fotoUrl?: string;
  activo: boolean;

export interface FiltroProductos {
  categoria?: Categoria;
  subcategoria?: string;
  uso?: Uso;
  presupuestoMax?: number;
  texto?: string;
  limite?: number;
}

const ORDEN_PRIORIDAD = { alta: 0, media: 1, baja: 2 } as const;
export const sinAcentos = (s: string): string => s.normalize("NFD").replace(/\p{Diacritic}/gu, "").toLowerCase();

export function filtrarProductos(productos: Producto[], f: FiltroProductos): Producto[] {
  return productos
    .filter((p) => p.activo)
    .filter((p) => !f.categoria || p.categoria === f.categoria)
    .filter((p) => !f.subcategoria || sinAcentos(p.subcategoria).includes(sinAcentos(f.subcategoria)))
    .filter((p) => !f.uso || p.uso.includes(f.uso))
    .filter((p) => f.presupuestoMax === undefined || p.precio <= f.presupuestoMax)
    .filter((p) => !f.texto || sinAcentos(p.nombre).includes(sinAcentos(f.texto)))
    .sort((a, b) => ORDEN_PRIORIDAD[a.prioridad] - ORDEN_PRIORIDAD[b.prioridad] || a.precio - b.precio)
    .slice(0, f.limite ?? 5);
}

export const formatoPrecio = (n: number): string => `$${n.toLocaleString("en-US")} + IVA`;
```

```ts
import { type Categoria, type Producto, sinAcentos, type Uso } from "./productos.js";

const OBLIGATORIAS = ["Categoria", "Subcategoria", "Producto", "Precio_MXN", "Stock", "Prioridad_Venta"] as const;

// Uso recomendado por subcategoría, tomado del cerebro de Tadeo.
const USO_POR_SUBCATEGORIA: Record<string, Uso[]> = {
  lineal: ["colaboradores", "administracion"],
  ejecutivo: ["direccion"],
  secretarial: ["recepcion", "administracion"],
  "semi ejecutivo": ["direccion", "administracion"],
  archivero: ["archivo", "administracion"],
  ejecutiva: ["direccion"],
  operativa: ["colaboradores", "administracion"],
  multiusos: ["visitas", "recepcion", "capacitacion"],
};

export const slug = (s: string): string =>
  sinAcentos(s)
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");

function categoria(cat: string, sub: string): Categoria {
  if (sinAcentos(cat).includes("silla")) return "silla";
  return sinAcentos(sub).includes("archivero") ? "archivero" : "escritorio";
}

function stock(s: string): Producto["stock"] {
  const n = sinAcentos(s);
  if (n.includes("proveedor")) return "proveedor";
  if (n.includes("alto")) return "alto";
  if (n.includes("medio")) return "medio";
  return "bajo";
}

const prioridad = (s: string): Producto["prioridad"] => {
  const n = sinAcentos(s);
  return n.includes("alta") ? "alta" : n.includes("media") ? "media" : "baja";
};

export function mapearHoja(filas: string[][]): { productos: Producto[]; errores: string[] } {
  const [encabezado = [], ...resto] = filas;
  const col = new Map(encabezado.map((h, i) => [h.trim(), i]));
  const faltan = OBLIGATORIAS.filter((h) => !col.has(h));
  if (faltan.length > 0) throw new Error(`Faltan columnas en la hoja: ${faltan.join(", ")}`);
  const leer = (fila: string[], h: string): string => {
    const i = col.get(h);
    return i === undefined ? "" : (fila[i] ?? "").trim();
  };

  const productos: Producto[] = [];
  const errores: string[] = [];
  for (const fila of resto) {
    const nombre = leer(fila, "Producto");
    if (!nombre) continue;
    const precio = Number(leer(fila, "Precio_MXN").replace(/[$,\s]/g, ""));
    if (!Number.isFinite(precio) || precio <= 0) {
      errores.push(`Precio inválido en «${nombre}»: ${leer(fila, "Precio_MXN")}`);
      continue;
    }
    const sub = leer(fila, "Subcategoria");
    const p: Producto = {
      slug: slug(nombre),
      categoria: categoria(leer(fila, "Categoria"), sub),
      subcategoria: sub,
      nombre,
      precio,
      stock: stock(leer(fila, "Stock")),
      prioridad: prioridad(leer(fila, "Prioridad_Venta")),
      uso: USO_POR_SUBCATEGORIA[sinAcentos(sub)] ?? [],
      activo: sinAcentos(leer(fila, "Activo")) !== "no",
    };
    const opcionales: [keyof Producto, string][] = [
      ["respuestaBase", "Respuesta_Chatbot_Base"],
      ["medidas", "Medidas"],
      ["colores", "Colores"],
      ["fotoUrl", "Foto_URL"],
    ];
    for (const [campo, h] of opcionales) {
      const v = leer(fila, h);
      if (v) Object.assign(p, { [campo]: v });
    }
    productos.push(p);
  }
  return { productos, errores };
}
```

```ts
  | { tipo: "bot" | "pausa" | "ganado"; ref?: string }
  | { tipo: "perdido"; ref?: string; motivo?: string }
  | { tipo: "ayuda" }
  | { tipo: "desconocido" };

export const AYUDA_COMANDOS = [
  "Comandos (mándalos a este número desde tu WhatsApp personal, o respondiendo a la alerta):",
  "/bot 8112345678  el asistente vuelve a atender ese chat",
  "/pausa 8112345678  el asistente se calla en ese chat hasta /bot",
  "/ganado 8112345678  venta cerrada",
  "/perdido 8112345678 motivo  venta perdida",
  "En Instagram usa el usuario: /bot @usuario",

function separarRef(resto: string): { ref?: string; libre: string } {
  const [primera = "", ...demas] = resto.split(/\s+/);
  if (primera.startsWith("@")) return { ref: primera, libre: demas.join(" ") };
  // Un teléfono puede venir con espacios: toma los dígitos iniciales.
  const m = resto.match(/^([\d\s+-]{10,})(.*)$/);
  if (m) {
    const digitos = (m[1] ?? "").replace(/\D/g, "");
    if (digitos.length >= 10) return { ref: digitos.slice(-10), libre: (m[2] ?? "").trim() };
  }
  return { libre: resto };
}

export function parsearComando(texto: string): Comando | null {
  const t = texto.trim();
  if (!t.startsWith("/")) return null;
  const [cabeza = "", ...cola] = t.slice(1).split(/\s+/);
  const nombre = cabeza.toLowerCase();
  const { ref, libre } = separarRef(cola.join(" ").trim());
  const conRef = <T extends object>(o: T) => (ref ? { ...o, ref } : o);
  switch (nombre) {
    case "bot":
    case "pausa":
    case "ganado":
      return conRef({ tipo: nombre });
    case "perdido":
      return conRef(libre ? { tipo: "perdido" as const, motivo: libre } : { tipo: "perdido" as const });
    case "ayuda":
      return { tipo: "ayuda" };
    default:
      return { tipo: "desconocido" };
  }
}
```

```ts
import { ajustarVentanaToques, horas, proximaA } from "../hora.js";

export interface EstadoSeguimiento {
  canal: Canal;
  modo: "bot" | "esperando_humano" | "humano" | "pausado";
  estado: string;
  toques: number;
  noContactar: boolean;
  ultimoCliente: Date;
  ultimoBot: Date;
  ventanaGratisHasta?: Date;
}

export interface Toque {
  tipo: "toque1" | "toque2" | "cierre";
  programadoPara: Date;
  usarPlantilla: boolean;
}

const ESTADOS_CON_SEGUIMIENTO = new Set(["nuevo", "calificado", "cotizado"]);

export function siguienteToque(s: EstadoSeguimiento): Toque | null {
  if (s.modo !== "bot" || s.noContactar || !ESTADOS_CON_SEGUIMIENTO.has(s.estado)) return null;
  const cierraVentana24 = s.ultimoCliente.getTime() + horas(24);

  if (s.toques === 0) {
    const cuando = ajustarVentanaToques(new Date(s.ultimoBot.getTime() + horas(3)));
    if (cuando.getTime() < cierraVentana24) return { tipo: "toque1", programadoPara: cuando, usarPlantilla: false };
    // La ventana se cerraría: se salta al toque 2.
    return siguienteToque({ ...s, toques: 1 });
  }

  if (s.toques === 1) {
    const cuando = proximaA(new Date(s.ultimoBot.getTime() + horas(12)), 10);
    const dentro24 = cuando.getTime() < cierraVentana24;
    const dentroGratis = s.ventanaGratisHasta !== undefined && cuando.getTime() < s.ventanaGratisHasta.getTime();
    if (s.canal === "instagram" && !dentro24) {
      return { tipo: "cierre", programadoPara: cuando, usarPlantilla: false };
    }
    return { tipo: "toque2", programadoPara: cuando, usarPlantilla: !dentro24 && !dentroGratis };
  }

  return { tipo: "cierre", programadoPara: new Date(s.ultimoBot.getTime() + horas(24)), usarPlantilla: false };
}
```

```ts
export class AlmacenSupabase implements Almacen {
  constructor(private readonly db: SupabaseClient, private readonly bucket = "adjuntos") {}
  async guardar(ruta: string, bytes: Uint8Array, mime: string) {
    const r = await this.db.storage.from(this.bucket).upload(ruta, bytes, { contentType: mime, upsert: true });
    if (r.error) throw new Error(`Storage: ${r.error.message}`);
  }
  async leer(ruta: string) {
    const r = await this.db.storage.from(this.bucket).download(ruta);
    if (r.error || !r.data) throw new Error(`Storage: ${r.error?.message ?? "sin datos"}`);
    return new Uint8Array(await r.data.arrayBuffer());
  }
}
```

```ts
import type Anthropic from "@anthropic-ai/sdk";
import type { Mensaje } from "../datos/tipos.js";

type Bloque = Anthropic.TextBlockParam | Anthropic.ImageBlockParam;
export type ImagenLista = { base64: string; mime: string };

const MIMES_IMAGEN = new Set(["image/jpeg", "image/png", "image/gif", "image/webp"]);

function bloques(m: Mensaje, imagenes: Map<string, ImagenLista>): Bloque[] {
  const salida: Bloque[] = [];
  if (m.tipo === "imagen") {
    const img = imagenes.get(m.id);
    if (img && MIMES_IMAGEN.has(img.mime)) {
      salida.push({ type: "image", source: { type: "base64", media_type: img.mime as "image/jpeg", data: img.base64 } });
    } else {
      salida.push({ type: "text", text: "[el cliente mandó una imagen que ya no está disponible]" });
    }
  }
  const texto = m.autor === "humano" ? `[Bryan escribió]: ${m.texto ?? ""}` : (m.texto ?? "");
  if (texto.trim()) salida.push({ type: "text", text: texto });
  return salida;
}

export function aMensajesClaude(mensajes: Mensaje[], imagenes: Map<string, ImagenLista>): Anthropic.MessageParam[] {
  const salida: { role: "user" | "assistant"; content: Bloque[] }[] = [];
  for (const m of mensajes) {
    if (m.autor === "sistema") continue;
    const rol = m.autor === "cliente" ? "user" : "assistant";
    if (salida.length === 0 && rol === "assistant") continue;
    const contenido = bloques(m, imagenes);
    if (contenido.length === 0) continue;
    const ultimo = salida[salida.length - 1];
    if (ultimo && ultimo.role === rol) ultimo.content.push(...contenido);
    else salida.push({ role: rol, content: contenido });
  }
  return salida;
}
```

```ts
/** Recorta a un máximo de bytes UTF-8 sin partir caracteres. */
export function recortarBytes(texto: string, max: number): string {
  if (Buffer.byteLength(texto) <= max) return texto;
  let s = texto;
  while (Buffer.byteLength(s) > max - 3) s = s.slice(0, -1);
  return `${s}...`;
}

export class ClienteInstagram implements Enviador {
  private readonly f: typeof fetch;
  private readonly base: string;

  }

      method: "POST",

      body: JSON.stringify(cuerpo),
    });
    const datos = (await r.json().catch(() => ({}))) as { message_id?: string; error?: { message?: string; code?: number } };
    if (!r.ok || datos.error) throw new Error(`Instagram ${r.status} ${datos.error?.code ?? ""}: ${datos.error?.message ?? "error"}`);
    return datos.message_id ?? "";
  }

  texto(para: string, texto: string) {
```

```ts
export type Transcriptor = (bytes: Uint8Array, mime: string) => Promise<string>;

export function transcriptorOpenAI(apiKey: string, f: typeof fetch = fetch): Transcriptor {
  return async (bytes, mime) => {
    const form = new FormData();
    form.set("model", "gpt-4o-mini-transcribe");
    form.set("language", "es");
    const extension = mime.includes("ogg") ? "ogg" : mime.includes("mp4") ? "mp4" : mime.includes("mpeg") ? "mp3" : "webm";
    form.set("file", new Blob([bytes], { type: mime }), `audio.${extension}`);
    const r = await f("https://api.openai.com/v1/audio/transcriptions", { method: "POST", headers: { Authorization: `Bearer ${apiKey}` }, body: form });
    if (!r.ok) throw new Error(`Transcripción falló: ${r.status}`);
    return ((await r.json()) as { text?: string }).text?.trim() ?? "";
  };
}
```

```ts
import Anthropic from "@anthropic-ai/sdk";
import { type ContextoHerramientas, DEFINICIONES, ejecutarHerramienta } from "./herramientas.js";

export interface ModeloCliente {
  crear(p: Anthropic.MessageCreateParamsNonStreaming): Promise<Anthropic.Message>;
}

export function modeloAnthropic(apiKey: string): ModeloCliente {
  const cliente = new Anthropic({ apiKey, timeout: 60_000, maxRetries: 2 });
  return { crear: (p) => cliente.messages.create(p) };
}

export interface ResultadoRespuesta {
  texto: string;
  herramientas: string[];
  uso: { entrada: number; salida: number; cache: number };
}

export interface EntradaResponder {
  modelo: ModeloCliente;
  model: string;
  sistemaEstatico: string;
  sistemaDinamico: string;
  mensajes: Anthropic.MessageParam[];
  ctx: ContextoHerramientas;
  maxIteraciones?: number;
}

function parametros(e: EntradaResponder, mensajes: Anthropic.MessageParam[]): Anthropic.MessageCreateParamsNonStreaming {
  return {
    model: e.model,
    max_tokens: 1500,
    system: [
      { type: "text", text: e.sistemaEstatico, cache_control: { type: "ephemeral", ttl: "1h" } },
      { type: "text", text: e.sistemaDinamico },
    ],
    tools: DEFINICIONES,
    messages: mensajes,
    // Sonnet 5.5: sin temperature ni tool_choice forzado. Esfuerzo bajo para respuestas cortas.
    output_config: { effort: "low" },
  } as Anthropic.MessageCreateParamsNonStreaming;
}

export async function responder(e: EntradaResponder): Promise<ResultadoRespuesta> {
  const mensajes = [...e.mensajes];
  const usadas: string[] = [];
  const uso = { entrada: 0, salida: 0, cache: 0 };
  const max = e.maxIteraciones ?? 6;

  for (let i = 0; i < max; i++) {
    const r = await e.modelo.crear(parametros(e, mensajes));
    uso.entrada += r.usage?.input_tokens ?? 0;
    uso.salida += r.usage?.output_tokens ?? 0;
    uso.cache += r.usage?.cache_read_input_tokens ?? 0;

    if (r.stop_reason !== "tool_use") {
      const texto = r.content
        .filter((b): b is Anthropic.TextBlock => b.type === "text")
        .map((b) => b.text)
        .join("\n")
        .trim();
      if (!texto) throw new Error("Respuesta vacía del modelo");
      return { texto, herramientas: usadas, uso };
    }

    // Se devuelve el contenido completo (incluidos bloques de pensamiento) tal como llegó.
    mensajes.push({ role: "assistant", content: r.content as Anthropic.ContentBlockParam[] });
    const resultados: Anthropic.ToolResultBlockParam[] = [];
    for (const b of r.content) {
      if (b.type !== "tool_use") continue;
      usadas.push(b.name);
      const { contenido, esError } = await ejecutarHerramienta(b.name, b.input, e.ctx);
      resultados.push({ type: "tool_result", tool_use_id: b.id, content: contenido, ...(esError ? { is_error: true } : {}) });
    }
    mensajes.push({ role: "user", content: resultados });
  }
  throw new Error(`El modelo excedió ${max} iteraciones de herramientas`);
}

/** Resume un hilo largo en 3 a 5 frases para abrir uno nuevo. Sin herramientas. */
export async function resumirHilo(modelo: ModeloCliente, model: string, transcripcion: string): Promise<string> {
  const r = await modelo.crear({
    model,
    max_tokens: 400,
    system: "Resume esta conversación de ventas de muebles en 3 a 5 frases en español: qué busca el cliente, cantidades, zona, precios ya dados y en qué quedaron. Solo hechos.",
    messages: [{ role: "user", content: transcripcion }],
  });
  return r.content
    .filter((b): b is Anthropic.TextBlock => b.type === "text")
    .map((b) => b.text)
    .join(" ")
    .trim();
}
```

```ts
export async function conReintentos<T>(fn: () => Promise<T>, d: Deps, intentos = 3): Promise<T> {
  let ultimo: unknown;
  for (let i = 0; i < intentos; i++) {
    try {
      return await fn();
    } catch (e) {
      ultimo = e;
      if (i < intentos - 1) await d.esperar(1000 * 2 ** i);
    }
  }
  throw ultimo;
}

export async function enviarTextoBot(conv: Conversacion, contacto: Contacto, texto: string, d: Deps): Promise<void> {

async function imagenesRecientes(mensajes: Mensaje[], d: Deps): Promise<Map<string, ImagenLista>> {

  let i = mensajes.length - 1;
```

```ts
/** Lo que corre el cron de Vercel cada minuto. Cada paso es independiente. */
export async function ejecutarMinuto(d: Deps): Promise<Record<string, number | boolean | string>> {
  const salida: Record<string, number | boolean | string> = {};
  const pasos: [string, () => Promise<number | boolean>][] = [
    ["eventos", () => reprocesarEventos(d)],
    ["turnos", () => despacharTurnos(d)],
    ["seguimientos", () => procesarSeguimientos(d)],
    ["modos", () => vencerModos(d)],
    ["alertas", () => reenviarAlertas(d)],
    ["resumen", () => resumenDiario(d)],
  ];
  for (const [nombre, paso] of pasos) {
    try {
      salida[nombre] = await paso();
    } catch (e) {
      salida[nombre] = `error: ${(e as Error).message}`;
    }
```

```ts
import { firmaValida } from "../canales/firma.js";
import { parsearInstagram } from "../canales/instagram.js";
import { type Canal, type EventoCanal, idEvento } from "../canales/tipos.js";
import { parsearWhatsApp } from "../canales/whatsapp.js";
import type { Deps } from "./deps.js";
import { procesarLote } from "./entrada.js";

export function verificarSuscripcion(req: Request, token: string): Response {
  const u = new URL(req.url);
  if (u.searchParams.get("hub.mode") === "subscribe" && u.searchParams.get("hub.verify_token") === token) {
    return new Response(u.searchParams.get("hub.challenge") ?? "", { status: 200 });
  }
  return new Response("forbidden", { status: 403 });
}

export function autorizado(req: Request, secreto: string): boolean {
  return req.headers.get("authorization") === `Bearer ${secreto}`;
}

export async function manejarWebhook(req: Request, canal: Canal, d: Deps, diferir: (p: Promise<unknown>) => void): Promise<Response> {
  const crudo = await req.text();
  if (!firmaValida(crudo, req.headers.get("x-hub-signature-256"), d.config.META_APP_SECRET)) {
    return new Response("firma inválida", { status: 403 });
  }
  let payload: unknown;
  try {
    payload = JSON.parse(crudo);
  } catch {
    return new Response("ok");
  }
  const eventos = canal === "whatsapp" ? parsearWhatsApp(payload) : parsearInstagram(payload, d.config.IG_USER_ID);
  const nuevos: EventoCanal[] = [];
  for (const ev of eventos) if (await d.repo.registrarEvento(idEvento(ev), canal, ev)) nuevos.push(ev);
  if (nuevos.length) diferir(procesarLote(nuevos, d));
  return new Response("ok");
}
```

```ts
export interface Hallazgo {
  regla: string;
  ok: boolean;
  detalle?: string;
}

const MONTOS_DE_REGLAS = new Set([200, 500, 40000]);
const aNumero = (s: string) => Number(s.replace(/[$,\s]/g, ""));

export function revisarTexto(texto: string, preciosReales: number[]): Hallazgo[] {
  const montos = [...texto.matchAll(/\$\s?[\d,]+(?:\.\d+)?/g)];
  const inventados: string[] = [];
  const sinIva: string[] = [];
  for (const m of montos) {
    const n = aNumero(m[0]);
    if (MONTOS_DE_REGLAS.has(n)) continue;
    const real = preciosReales.some((p) => n === p || (n % p === 0 && n / p <= 100));
    if (!real) inventados.push(m[0]);
    const despues = texto.slice((m.index ?? 0) + m[0].length, (m.index ?? 0) + m[0].length + 20);
    if (!/IVA/.test(despues)) sinIva.push(m[0]);
  }
  return [
    { regla: "sin_exclamacion", ok: !/[!¡]/.test(texto) },
    { regla: "max_dos_preguntas", ok: (texto.match(/\?/g)?.length ?? 0) <= 2 },
    { regla: "longitud", ok: texto.length <= 700, detalle: `${texto.length} caracteres` },
    { regla: "precio_real", ok: inventados.length === 0, detalle: inventados.join(", ") },
    { regla: "iva_junto_al_precio", ok: sinIva.length === 0, detalle: sinIva.join(", ") },
  ];
}
```

```ts
import { mkdir, writeFile } from "node:fs/promises";
import { join } from "node:path";
import { modeloAnthropic } from "../src/cerebro/responder.js";
import { procesarEvento } from "../src/flujo/entrada.js";
import { procesarTurno } from "../src/flujo/turno.js";
import { aLocal } from "../src/hora.js";
import { mapearHoja } from "../src/negocio/hoja.js";
import { ESCENARIOS, type Escenario } from "./escenarios.js";
import { ENCABEZADO, FILAS } from "./inventario.js";
import { type Hallazgo, revisarTexto } from "./revisar.js";
import { depsDePrueba, mensajeWa } from "./unitarias/apoyo.js";

const CLIENTE = "5218112345678";

async function correr(e: Escenario) {
  const apiKey = process.env.ANTHROPIC_API_KEY;
  if (!apiKey) throw new Error("Falta ANTHROPIC_API_KEY");
  const { deps, repo, wa, reloj } = depsDePrueba({ modelo: modeloAnthropic(apiKey) });
  if (process.env.CLAUDE_MODEL) deps.config = { ...deps.config, CLAUDE_MODEL: process.env.CLAUDE_MODEL };
  const { productos } = mapearHoja([ENCABEZADO, ...FILAS]);
  await repo.sincronizarProductos(productos);
  if (e.inicio) reloj.fijar(aLocal(...e.inicio));

  const transcripcion: string[] = [`# ${e.nombre}`, ""];
  for (const [i, texto] of e.mensajes.entries()) {
    transcripcion.push(`**Cliente:** ${texto}`, "");
    const antes = wa.enviados.length;
    await procesarEvento(mensajeWa(texto, { fecha: deps.ahora().toISOString(), ...(i === 0 && e.referencia ? { referencia: e.referencia } : {}) }), deps);
    const conv = await repo.conversacionDeContacto((await repo.contactoPorExterno("whatsapp", CLIENTE))?.id ?? "");
    if (conv) await procesarTurno(conv.id, deps);
    for (const env of wa.enviados.slice(antes)) {
      if (env.tipo === "escribiendo") continue;
      const quien = env.para === CLIENTE ? "Asistente" : "Alerta a dueño";
      transcripcion.push(`**${quien}** (${env.tipo}): ${env.texto ?? env.plantilla ?? env.url ?? ""}`, "");
    }
    reloj.avanzar(60_000);
  }

  const contacto = await repo.contactoPorExterno("whatsapp", CLIENTE);
  const conv = await repo.conversacionDeContacto(contacto?.id ?? "");
  const lead = await repo.obtenerLead(conv?.id ?? "");
  const textosBot = wa.textos(CLIENTE);
  const herramientas = repo.llamadas.flatMap((l) => l.herramientas);
  const precios = productos.map((p) => p.precio);

  const hallazgos: Hallazgo[] = textosBot.flatMap((t, i) => revisarTexto(t, precios).map((h) => ({ ...h, regla: `${h.regla}#${i + 1}` })));
  if (e.espera.escalar) hallazgos.push({ regla: "escala_con_motivo", ok: lead.motivoEscalacion === e.espera.escalar, detalle: `esperado ${e.espera.escalar}, obtenido ${lead.motivoEscalacion ?? "ninguno"}` });
  if (e.espera.sinEscalar) hallazgos.push({ regla: "no_escala", ok: !lead.motivoEscalacion, detalle: lead.motivoEscalacion ?? "" });
  if (e.espera.marcador) hallazgos.push({ regla: "marcador", ok: lead.motivosPrioridad.includes(e.espera.marcador), detalle: lead.motivosPrioridad.join(", ") });
  for (const h of e.espera.herramientas ?? []) hallazgos.push({ regla: `usa_${h}`, ok: herramientas.includes(h) });
  if (e.espera.textoIncluye) hallazgos.push({ regla: "texto_incluye", ok: textosBot.some((t) => t.toLowerCase().includes(e.espera.textoIncluye as string)) });
  if (lead.nombre || lead.direccion) hallazgos.push({ regla: "aviso_privacidad", ok: lead.avisoEnviado && textosBot.some((t) => t.includes(deps.config.AVISO_PRIVACIDAD_URL)) });

  return { transcripcion: transcripcion.join("\n"), hallazgos };
}

async function main() {
  const filtro = process.argv[2];
  const lista = ESCENARIOS.filter((e) => !filtro || e.nombre === filtro);
  const carpeta = join("pruebas", "transcripciones", new Date().toISOString().slice(0, 16).replace(/[:T]/g, "-"));
  await mkdir(carpeta, { recursive: true });
  let fallidos = 0;
  for (const e of lista) {
    const { transcripcion, hallazgos } = await correr(e);
    const malos = hallazgos.filter((h) => !h.ok);
    await writeFile(join(carpeta, `${e.nombre}.md`), `${transcripcion}\n## Rúbricas\n\n${hallazgos.map((h) => `- ${h.ok ? "ok" : "FALLA"} ${h.regla} ${h.detalle ?? ""}`).join("\n")}\n`);
    console.log(`${malos.length ? "FALLA" : "ok   "} ${e.nombre}${malos.length ? `: ${malos.map((h) => h.regla).join(", ")}` : ""}`);
    if (malos.length) fallidos++;
  }
  console.log(`\n${lista.length - fallidos}/${lista.length} escenarios pasan. Transcripciones en ${carpeta}`);
  process.exitCode = fallidos ? 1 : 0;
}

await main();
```
